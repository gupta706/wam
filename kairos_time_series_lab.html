# PyTorch Observability Gramian & Fisher Information Inspector import torch def compute_observability_gramian(A, C, T_steps=50, dt=0.02): # A: [n, n], C: [m, n] n = A.size(0) Wo = torch.zeros(n, n, device=A.device) expAt = torch.eye(n, device=A.device) dA = torch.matrix_exp(A * dt) for _ in range(T_steps): integrand = expAt.T @ C.T @ C @ expAt Wo += integrand * dt expAt = expAt @ dA eigvals, _ = torch.linalg.eigh(Wo) cond_number = eigvals[-1] / torch.clamp(eigvals[0], min=1e-7) return Wo, eigvals, cond_number
